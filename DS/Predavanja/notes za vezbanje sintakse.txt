//IDL proizvoljni primer
//========================IDL fajl, kalk.x================

struct operandi{
	int x,
	int y
}

program KALKULATOR {	//ime programa

	version KALKULATOR_VERSION { 	//ime verzije
		
		int pomnozi(operandi) = 1;
		int kvadriraj(int x) = 2;
	
	
	} = 1	//broj verzije

} = 0x03ffffff	//broj programa na na osnovu koga se zna da je taj (kao id)

//Kompajliranjem se dobija:

rpcgen -C kalk.x

->
	kalk.h //header fajl
	kalk.clnt.c //klijent stub
	kalk.svc.c //server stub
	
//Klijent poziva program sa
(imeprocedure_brojVerzije)
-> 

//==================Klijent========================

#include <stdio.h>
#include <rpc/rpc.h>
#include <kalk.h>

int main(int argc, char* argv[])
{
	CLIENT *clnt;	//RPC handler
	char* server;
	int *proizvod, *kvadrat;
	operandi op;	//struktura definisana u IDL fajlu
	
	server = argv[1]
	clnt = cln_create(server, KALKULATOR_VERSION, "udp")
	
	if(clnt == NULL) exit(2);
	
	op.x = atoi(argv[2]);
	op.y = atoi(argv[3]);
	
	zbir = pomnozi_1(&op, clnt)
	
	if(zbir == (int* NULL)) exit(3);
	
	kvadrat = kvadriraj_1(2);
	
	if(kvadrat == (int* NULL)) exit(4);
	
	clnt_destroy(clnt);
	exit(0);
	
}



//==========Kako se linkuju klijent i server========
primer.x

rpcgen -C primer.x		(primer.h, primer_clnt.c, primer_svc.c)

// kompajliranje klijenta(servera) i linkovanje sa klijent(server) stubom
cc -o klijent klijent.c primer_clnt.c
cc -o server server.c primer_svc.c
